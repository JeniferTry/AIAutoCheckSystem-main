1、查找进程、杀死进程
# 列出全部进程
tasklist

# 过滤查找指定进程（示例查找chrome）
tasklist | findstr chrome

/F = 强制终止（卡死无响应必须加）
/T = 连带子进程一起杀掉（浏览器、软件很常用）
/PID：按进程 ID 杀（只杀这一个）
/IM：按程序名杀（同名所有进程全部关闭）

# 强制关闭所有记事本
taskkill /F /IM notepad.exe

# 强制关闭所有chrome，连同子进程
taskkill /F /T /IM chrome.exe

按 PID 杀死（精准单个进程）
taskkill /F /PID 12480
